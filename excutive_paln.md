3.5 SQL Views الجاهزة (الأولوية القصوى)
sql-- ── ميزان المراجعة ──────────────────────────────────────────
CREATE VIEW trial_balance AS
SELECT
  jel.account_code,
  coa.name          AS account_name,
  coa.account_type,
  SUM(jel.debit)    AS total_debit,
  SUM(jel.credit)   AS total_credit,
  SUM(jel.debit) - SUM(jel.credit) AS net_balance
FROM journal_entry_lines jel
JOIN journal_entries     je  ON je.id = jel.entry_id
JOIN chart_of_accounts   coa ON coa.code = jel.account_code
                             AND coa.company_id = je.company_id
WHERE je.is_posted = 1
GROUP BY jel.account_code, coa.name, coa.account_type;

-- ── قائمة الدخل (P&L) ──────────────────────────────────────
CREATE VIEW profit_and_loss AS
SELECT
  coa.account_type,
  coa.code,
  coa.name,
  SUM(jel.credit) - SUM(jel.debit) AS balance
FROM journal_entry_lines jel
JOIN journal_entries   je  ON je.id = jel.entry_id
JOIN chart_of_accounts coa ON coa.code = jel.account_code
                           AND coa.company_id = je.company_id
WHERE je.is_posted = 1
  AND coa.account_type IN ('revenue','expense')
GROUP BY coa.account_type, coa.code, coa.name;

-- ── تكلفة الفدان (Cost per Feddan) ─────────────────────────
CREATE VIEW cost_per_feddan AS
SELECT
  f.id            AS field_id,
  f.name          AS field_name,
  f.area_feddan,
  f.crop_type,
  s.name          AS season_name,
  COALESCE(SUM(wt.total_cost), 0) AS total_work_cost,
  COALESCE(inv.total_inv_value, 0) AS total_inv_cost,
  (COALESCE(SUM(wt.total_cost), 0) + COALESCE(inv.total_inv_value, 0))
    / NULLIF(f.area_feddan, 0)     AS cost_per_feddan
FROM fields f
LEFT JOIN seasons s ON s.id = f.season_id
LEFT JOIN work_orders wo ON wo.field_id = f.id
LEFT JOIN work_tasks  wt ON wt.work_order_id = wo.id
LEFT JOIN (
  SELECT field_id, SUM(value_in) AS total_inv_value
  FROM inventory_movements
  WHERE movement_type = 'اضافة'
  GROUP BY field_id
) inv ON inv.field_id = f.id
GROUP BY f.id, f.name, f.area_feddan, f.crop_type, s.name, inv.total_inv_value;

8. نقاط التفوق التنافسي — ما يجعلنا مختلفين

هذه الميزات غير موجودة في المنافس ويجب أن تكون نقاط البيع الرئيسية:


Agricultural Seasons — كل شيء مربوط بموسم زراعي، لا يوجد في أي ERP عام
Offline-First Mobile — العمل في الحقل بدون إنترنت ومزامنة لاحقاً
Cost per Feddan — تقرير تكلفة الفدان وربحية كل قطعة بشكل تلقائي
Package-level Inventory — تتبع العبوة/السعة/العدد للمواد الزراعية
Cloudflare Edge — سرعة عالمية + لا سيرفر تقليدي + تكلفة منخفضة
Multi-tenant من اليوم الأول — بيع لشركات متعددة بنفس الـ codebase
Field Profitability — ربحية كل قطعة = مبيعات العقود - التكاليف الكاملة
